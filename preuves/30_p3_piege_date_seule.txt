"""M30 - piege data « cible sur la date seule » (lecture seule, aucune modification de la lib).

On construit la cible `panne` deux fois avec la MEME fonction build_dataset :
  - CORRECT : incident_ts = date + time        (ce que fait load_incidents)
  - PIEGE   : incident_ts = date seule (minuit) (l'erreur a ne pas faire)
puis on compare taux de panne et lignes etiquetees.
"""

from pathlib import Path

import pandas as pd

from indusense.config import settings
from indusense.data.loaders import build_dataset, load_incidents, load_pressure, load_temperature

raw = Path("data/raw")
temp = load_temperature(raw / "capteurs_temperature.csv")
pres = load_pressure(raw / "capteurs_pression.tsv")
inc_ok = load_incidents(raw / "releves_incidents.csv")

# PIEGE : on remplace l'horodatage par la date seule -> heure a 00:00.
inc_piege = inc_ok.copy()
inc_piege["incident_ts"] = inc_piege["incident_ts"].dt.normalize()

w = settings.incident_window_hours
ok = build_dataset(temp, pres, inc_ok, window_hours=w)
piege = build_dataset(temp, pres, inc_piege, window_hours=w)

print("1) SYMPTOME - taux de panne (meme population : jointure, 1920 lignes)")
for nom, df in (("date + time (correct)", ok), ("date seule  (piege)  ", piege)):
    print(f"   {nom} : panne_count={int(df['panne'].sum())} panne_rate={df['panne'].mean():.8f}")

print("\n2) DIAGNOSTIC - decalage de chaque incident (heure perdue = recul de la fenetre)")
for (_, a), (_, b) in zip(inc_ok.iterrows(), inc_piege.iterrows(), strict=True):
    recul = (a["incident_ts"] - b["incident_ts"]).total_seconds() / 3600
    print(f"   {a['machine']} incident {a['incident_ts']} -> {b['incident_ts']}  (fenetre reculee de {recul:.0f} h)")

diff = (ok["panne"] != piege["panne"])
faux_pos = int(((piege["panne"] == 1) & (ok["panne"] == 0)).sum())
faux_neg = int(((piege["panne"] == 0) & (ok["panne"] == 1)).sum())
print("\n3) IMPACT - lignes dont l'etiquette change")
print(f"   lignes differentes : {int(diff.sum())} / {len(ok)}")
print(f"   etiquetees panne A TORT (heures saines, la veille) : {faux_pos}")
print(f"   pannes MANQUEES (heures juste avant l'incident)    : {faux_neg}")

print("\n4) CORRECTION - load_incidents combine deja date + time :")
print("   pd.to_datetime(df['date'].astype(str) + ' ' + df['time'].astype(str))  (loaders.py:93)")
