"""Demo M23 TP2 : moyenne glissante avec et sans shift(1) (aucun fichier du projet modifie)."""

import pandas as pd

from indusense.features.temporal import add_temporal_features

df = pd.DataFrame(
    {
        "machine": ["MACH-01"] * 4,
        "timestamp": pd.date_range("2025-01-01", periods=4, freq="h"),
        "temperature": [10.0, 20.0, 30.0, 40.0],
        "pressure_bar": [195.0, 196.0, 197.0, 198.0],
    }
)

out = add_temporal_features(df, value_cols=("temperature",), lags=(1,), windows=(2,))
out["SANS_shift_roll2_FUITE"] = df.groupby("machine")["temperature"].transform(
    lambda s: s.rolling(2).mean()
)
cols = ["timestamp", "temperature", "temperature_lag1", "temperature_roll2_mean",
        "SANS_shift_roll2_FUITE"]
print(out[cols].to_string(index=False))
print()
print("Ligne 3 (temperature=30) :")
print("  avec shift(1) -> moyenne(10, 20) =", out.loc[2, "temperature_roll2_mean"])
print("  sans shift(1) -> moyenne(20, 30) =", out.loc[2, "SANS_shift_roll2_FUITE"],
      "<- contient la valeur courante 30 : FUITE")
