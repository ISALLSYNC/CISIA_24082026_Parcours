"""M29 - piege data : fuite inter-machines dans merge_asof (aucun fichier du projet modifie)."""

import pandas as pd

from indusense.config import settings
from indusense.data.loaders import load_pressure, load_temperature

KW = dict(on="timestamp", tolerance=pd.Timedelta("90min"), direction="nearest")

print("=== 1) PETIT EXEMPLE (fiche TD 29) ===")
ts = pd.to_datetime(["2026-01-01 00:00", "2026-01-01 00:01"])
temp = pd.DataFrame({"timestamp": ts, "machine": ["MACH-01", "MACH-02"], "temp_c": [40.0, 80.0]})
pres = pd.DataFrame({"timestamp": ts, "machine": ["MACH-02", "MACH-01"], "pressure_bar": [2.0, 9.0]})
avec = pd.merge_asof(temp.sort_values("timestamp"), pres.sort_values("timestamp"), by="machine", **KW)
sans = pd.merge_asof(
    temp.sort_values("timestamp"),
    pres.rename(columns={"machine": "machine_de_la_pression"}).sort_values("timestamp"),
    **KW,
)
print("AVEC by='machine' :")
print(avec[["timestamp", "machine", "temp_c", "pressure_bar"]].to_string(index=False))
print("SANS by='machine' (on garde la machine d'ou vient la pression pour la montrer) :")
print(sans[["timestamp", "machine", "temp_c", "pressure_bar", "machine_de_la_pression"]].to_string(index=False))
print("-> MACH-01 recoit", sans.loc[sans.machine == "MACH-01", "pressure_bar"].item(),
      "bar SANS by (pression de", sans.loc[sans.machine == "MACH-01", "machine_de_la_pression"].item(),
      ") contre", avec.loc[avec.machine == "MACH-01", "pressure_bar"].item(), "bar AVEC by.")

print()
print("=== 2) VRAIES DONNEES InduSense (data/raw, chargees par les loaders du projet) ===")
d = settings.data_dir
t = load_temperature(d / "capteurs_temperature.csv").sort_values("timestamp")
p = load_pressure(d / "capteurs_pression.tsv").sort_values("timestamp")
print(f"releves temperature : {len(t)} | releves pression : {len(p)} | machines : {sorted(t.machine.unique())}")

j_avec = pd.merge_asof(t, p, by="machine", **KW)
non_joint = j_avec["pressure_bar"].isna().sum()
print(f"AVEC by='machine' : {non_joint} lignes sans pression a +/-90 min sur {len(j_avec)} "
      f"-> residu non-joint = {non_joint / len(j_avec):.2%}")

j_sans = pd.merge_asof(t, p.rename(columns={"machine": "machine_de_la_pression"}), **KW)
jointes = j_sans.dropna(subset=["pressure_bar"])
fuite = (jointes["machine"] != jointes["machine_de_la_pression"]).sum()
print(f"SANS by='machine' : {j_sans['pressure_bar'].isna().sum()} lignes sans pression ; "
      f"{fuite} lignes sur {len(jointes)} jointes ({fuite / len(jointes):.2%}) recoivent la pression "
      f"d'une AUTRE machine -> AUCUNE erreur levee")
print("exemples de lignes contaminees :")
ex = jointes[jointes["machine"] != jointes["machine_de_la_pression"]].head(4)
print(ex[["timestamp", "machine", "temperature", "pressure_bar", "machine_de_la_pression"]].to_string(index=False))
